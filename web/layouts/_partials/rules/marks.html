{{- /* A node's marks at the end of its row in the list of the rules, in four places: its status; whether the defaults enable it, a dot filled or a ring, empty when that depends on another source; and its uses, instructions and review, each empty when the node has not got it (a one-of has those of its options). Each mark is named for assistive technology and on hover. Takes the node's data; returns the HTML. */ -}} {{- $marks := partial "rules/status.html" .status }} {{- if eq .onByDefault true }} {{- $marks = printf `%s` $marks (i18n "default-on") (i18n "default-on") }} {{- else if eq .onByDefault false }} {{- $marks = printf `%s` $marks (i18n "default-off") (i18n "default-off") }} {{- else }} {{- $marks = printf `%s` $marks }} {{- end }} {{- $uses := or .optionUses .uses (slice) }} {{- range $use, $icon := dict "instructions" "scroll-text" "review" "scan-eye" }} {{- if in $uses $use }} {{- $name := i18n (printf "uses-%s" $use) }} {{- $marks = printf `%s%s` $marks $name $name (partial "icon.html" (printf "lucide:%s" $icon)) }} {{- else }} {{- $marks = printf `%s` $marks }} {{- end }} {{- end }} {{- return printf `%s` $marks }}
